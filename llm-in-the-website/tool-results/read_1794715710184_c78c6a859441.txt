    33→   742→   742→  // Each CJK character width ≈ pt × 20 twips
    34→   743→   743→  const charWidth = (pt) => pt * 20;
    35→   744→   744→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / charWidth(pt));
    36→   745→   745→
    37→   746→   746→  // Try from preferredPt downward until title fits in ≤ 3 lines
    38→   747→   747→  let titlePt = preferredPt;
    39→   748→   748→  let lines;
    40→   749→   749→  while (titlePt >= minPt) {
    41→   750→   750→    const cpl = charsPerLine(titlePt);
    42→   751→   751→    if (cpl < 2) { titlePt -= 2; continue; }
    43→   752→   752→    lines = splitTitleLines(title, cpl);
    44→   753→   753→    if (lines.length <= 3) break;
    45→   754→   754→    titlePt -= 2;
    46→   755→   755→  }
    47→   756→   756→
    48→   757→   757→  // If still > 3 lines at minPt, force 3 lines
    49→   758→   758→  if (!lines || lines.length > 3) {
    50→   759→   759→    const cpl = charsPerLine(minPt);
    51→   760→   760→    lines = splitTitleLines(title, cpl);
    52→   761→   761→    titlePt = minPt;
    53→   762→   762→  }
    54→   763→   763→
    55→   764→   764→  return { titlePt, titleLines: lines };
    56→   765→   765→}
    57→   766→   766→
    58→   767→   767→/**
    59→   768→   768→ * Smart Chinese title line-breaking — breaks at semantic boundaries, never mid-word.
    60→   769→   769→ *
    61→   770→   770→ * Rules:
    62→   771→   771→ * 1. Prefer breaking after particles, punctuation, connectors, underscores, spaces
    63→   772→   772→ * 2. Never split a compound word (e.g., "管理规范" must not become "管理规" + "范")
    64→   773→   773→ * 3. No single-character orphan on the last line — merge into previous line
    65→   774→   774→ * 4. If no good break point found within 60-130% of charsPerLine, break at charsPerLine
    66→   775→   775→ *
    67→   776→   776→ * @param {string} title        Full title string
    68→   777→   777→ * @param {number} charsPerLine Max characters per line at current font size
    69→   778→   778→ * @returns {string[]}          Array of line strings
    70→   779→   779→ */
    71→   780→   780→function splitTitleLines(title, charsPerLine) {
    72→   781→   781→  if (title.length <= charsPerLine) return [title];
    73→   782→   782→
    74→   783→   783→  // Characters that are safe break points (break AFTER these)
    75→   784→   784→  const breakAfter = new Set([
    76→   785→   785→    ...'，。、；：！？',              // CJK punctuation
    77→   786→   786→    ...'的与和及之在于为',            // CJK particles/prepositions
    78→   787→   787→    ...'-_—–·/',                     // connectors
    79→   788→   788→    ...' \t',                         // whitespace
    80→   789→   789→  ]);
    81→   790→   790→
    82→   791→   791→  const lines = [];
    83→   792→   792→  let remaining = title;
    84→   793→   793→
    85→   794→   794→  while (remaining.length > charsPerLine) {
    86→   795→   795→    let breakAt = -1;
    87→   796→   796→
    88→   797→   797→    // Search backward from charsPerLine to 60% for a break point
    89→   798→   798→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    90→   799→   799→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) {
    91→   800→   800→        breakAt = i;
    92→   801→   801→        break;
    93→   802→   802→      }
    94→   803→   803→    }
    95→   804→   804→
    96→   805→   805→    // If not found, search forward up to 130%
    97→   806→   806→    if (breakAt === -1) {
    98→   807→   807→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    99→   808→   808→      for (let i = charsPerLine + 1; i < limit; i++) {
   100→   809→   809→        if (breakAfter.has(remaining[i - 1])) {
   101→   810→   810→          breakAt = i;
   102→   811→   811→          break;
   103→   812→   812→        }
   104→   813→   813→      }
   105→   814→   814→    }
   106→   815→   815→
   107→   816→   816→    // Last resort: break at charsPerLine, but avoid splitting compound CJK words
   108→   817→   817→    if (breakAt === -1) {
   109→   818→   818→      breakAt = charsPerLine;
   110→   819→   819→      // If both chars at the break boundary are CJK (likely a compound word),
   111→   820→   820→      // step back 1 char to keep the word together
   112→   821→   821→      const prevChar = remaining[breakAt - 1];
   113→   822→   822→      const nextChar = remaining[breakAt];
   114→   823→   823→      if (prevChar && nextChar &&
   115→   824→   824→          !breakAfter.has(prevChar) && !breakAfter.has(nextChar) &&
   116→   825→   825→          /[\u4e00-\u9fff]/.test(prevChar) && /[\u4e00-\u9fff]/.test(nextChar)) {
   117→   826→   826→        breakAt = breakAt - 1;
   118→   827→   827→      }
   119→   828→   828→    }
   120→   829→   829→
   121→   830→   830→    lines.push(remaining.slice(0, breakAt).trim());
   122→   831→   831→    remaining = remaining.slice(breakAt).trim();
   123→   832→   832→  }
   124→   833→   833→  if (remaining) lines.push(remaining);
   125→   834→   834→
   126→   835→   835→  // Prevent single-character orphan on last line — merge into previous
   127→   836→   836→  if (lines.length > 1 && lines[lines.length - 1].length <= 2) {
   128→   837→   837→    const last = lines.pop();
   129→   838→   838→    lines[lines.length - 1] += last;
   130→   839→   839→  }
   131→   840→   840→
   132→   841→   841→  return lines;
   133→   842→   842→}
   134→   843→   843→
   135→   844→   844→/**
   136→   845→   845→ * Calculate dynamic spacing values for cover elements to fit within page height.
   137→   846→   846→ *
   138→   847→   847→ * @param {object} params
   139→   848→   848→ * @param {number} params.titleLineCount   Number of title lines
   140→   849→   849→ * @param {number} params.titlePt          Title font size in pt
   141→   850→   850→ * @param {boolean} params.hasSubtitle     Whether subtitle exists
   142→   851→   851→ * @param {boolean} params.hasEnglishLabel Whether English label exists
   143→   852→   852→ * @param {number} params.metaLineCount    Number of meta info lines
   144→   853→   853→ * @param {number} params.fixedHeight      Sum of fixed-height elements (color strips, accent bars, footer) in twips
   145→   854→   854→ * @param {number} params.pageHeight       Total page height in twips (default 16838)
   146→   855→   855→ * @returns {{ topSpacing, midSpacing, bottomSpacing }}  Spacing values in twips
   147→   856→   856→ */
   148→   857→   857→function calcCoverSpacing(params) {
   149→   858→   858→  const {
   150→   859→   859→    titleLineCount = 1, titlePt = 36, hasSubtitle = false,
   151→   860→   860→    hasEnglishLabel = false, metaLineCount = 0,
   152→   861→   861→    fixedHeight = 800, pageHeight = 16838,
   153→   862→   862→    marginTop = 0, marginBottom = 0,   // ★ NEW: pass actual section margins
   154→   863→   863→  } = params;
   155→   864→   864→
   156→   865→   865→  // ★ Safety margin: 1200 twips (cross-engine: MS Office renders large fonts
   157→   866→   866→  // taller than calculated; extra 400tw buffer prevents footer clipping)
   158→   867→   867→  const SAFETY = 1200;
   159→   868→   868→  // ★ Subtract page margins from available height (cover section may have margins)
   160→   869→   869→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
   161→   870→   870→
   162→   871→   871→  // ★ Accurate height estimation per element:
   163→   872→   872→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
   164→   873→   873→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
   165→   874→   874→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
   166→   875→   875→  const metaHeight = metaLineCount * (10 * 23 + 100);
   167→   876→   876→
   168→   877→   877→  // ★ Account for implicit paragraph heights:
   169→   878→   878→  const implicitParaHeight = 3 * 300;
   170→   879→   879→
   171→   880→   880→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight +
   172→   881→   881→                        metaHeight + fixedHeight + implicitParaHeight;
   173→   882→   882→
   174→   883→   883→  const remainingSpace = usableHeight - contentHeight;
   175→   884→   884→  const safeRemaining = Math.max(remainingSpace, 400);
   176→   885→   885→
   177→   886→   886→  // ★ Footer protection: bottomSpacing must be ≥ FOOTER_MIN to prevent
   178→   887→   887→  // footer + accent line from being clipped at the cell bottom edge
   179→   888→   888→  const FOOTER_MIN = 800;
   180→   889→   889→  const rawTop = Math.floor(safeRemaining * 0.45);
   181→   890→   890→  const rawBottom = Math.floor(safeRemaining * 0.45);
   182→   891→   891→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
   183→   892→   892→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   184→   893→   893→  const midSpacing = Math.max(safeRemaining - topSpacing - bottomSpacing, 0);
   185→   894→   894→
   186→   895→   895→  return { topSpacing, midSpacing, bottomSpacing };
   187→   896→   896→}
   188→   897→   897→```

   217→   926→   926→### ⚠️ CRITICAL — Cover Section Non-Negotiables (ALL Recipes)
   218→   927→   927→
   219→   928→   928→These 3 properties are MANDATORY for every cover implementation (R1–R7). Omitting ANY of them causes cover layout failure:
   220→   929→   929→
   221→   930→   930→1. **Cover section margin = 0**: The cover MUST be in its own section with `page.margin: { top: 0, bottom: 0, left: 0, right: 0 }`. Non-zero margins shrink the wrapper away from page edges → white gaps around the cover ("cover not filling the page"). This is the #1 cause of broken cover layouts.
   222→   931→   931→
   223→   932→   932→2. **Wrapper row exact height**: The outer wrapper table row MUST set `height: { value: 16838, rule: "exact" }`. Without this, content overflow pushes to page 2, or insufficient content leaves bottom whitespace.
   224→   933→   933→
   225→   934→   934→3. **Wrapper table borders = allNoBorders**: MUST explicitly set `borders: allNoBorders`. Default docx-js borders add ~8 twips per edge. MS Office includes border thickness in exact-height calculation → total exceeds 16838 → blank page 2 (WPS is lenient, MS Office is strict).
   226→   935→   935→
   227→   936→   936→**Cover section template (copy this for every Recipe):**
   228→   937→   937→```js
   229→   938→   938→sections: [

   232→   941→   941→    properties: {
   233→   942→   942→      page: {
   234→   943→   943→        size: { width: 11906, height: 16838 },
   235→   944→   944→        margin: { top: 0, bottom: 0, left: 0, right: 0 },

   237→   946→   946→    },
